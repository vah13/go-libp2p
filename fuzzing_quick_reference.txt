TESTING PATTERNS
================================================================================

PATTERN A - Simple Bytes Input
  func FuzzXXX(f *testing.F) {
      f.Fuzz(func(t *testing.T, data []byte) {
          result, err := FunctionUnderTest(data)
          // Verify no panic, handle error gracefully
      })
  }

PATTERN B - Protobuf Input
  func FuzzXXX(f *testing.F) {
      f.Fuzz(func(t *testing.T, data []byte) {
          msg := &pb.SomeMessage{}
          if err := proto.Unmarshal(data, msg); err == nil {
              FunctionUnderTest(msg)
          }
      })
  }

PATTERN C - Multi-input State Machine
  func FuzzXXX(f *testing.F) {
      f.Fuzz(func(t *testing.T, input1, input2 []byte) {
          state := setupState(input1)
          result, err := StateMachineFunc(state, input2)
          verifyStateInvariant(state)
      })
  }

PATTERN D - Property-Based (Roundtrip)
  func FuzzXXX(f *testing.F) {
      f.Fuzz(func(t *testing.T, data []byte) {
          obj, err := Unmarshal(data)
          if err == nil {
              marshaled, _ := Marshal(obj)
              obj2, _ := Unmarshal(marshaled)
              // Verify obj == obj2
          }
      })
  }
